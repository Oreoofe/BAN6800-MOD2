name: ci

on:
  pull_request:
    branches: [main]

jobs:
  test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: '3.11'
      - name: Install dependencies
        run: |
          pip install -r requirements.txt
      - name: Run fairness & security tests
        run: |
          pytest tests/ -v --tb=short
      - name: Verify no data files committed
        run: |
          if git ls-files | grep -E '\.(csv|parquet|xlsx)$|data/raw|data/processed'; then
            echo "DATA FILE COMMITTED — violation of no-PII rule"; exit 1;
          fi
